2) Add a base-safe asset helper

Create src/utils/asset.ts:

export function asset(p: string) {
  const base = import.meta.env.BASE_URL || "/";
  const clean = p.replace(/^\/+/, "");
  return base.endsWith("/") ? base + clean : base + "/" + clean;
}

3) Use the helper for viewer, worker, and file URL

In the component that renders the PDF iframe (e.g., PdfModal.tsx):

import { asset } from "@/utils/asset";
import { GlobalWorkerOptions } from "pdfjs-dist";

GlobalWorkerOptions.workerSrc = asset("pdfjs/build/pdf.worker.min.js");

const viewerUrl = asset("pdfjs/web/viewer.html");
// pub.pdfPath must look like "publications/hybrid-urbanism.pdf"
const fileUrl = asset(pub.pdfPath.replace(/^\/+/, ""));

// Be resilient to hosts that dislike Range requests in prod
const hash = import.meta.env.DEV
  ? "#zoom=page-fit"
  : "#zoom=page-fit&disableRange=true";

<iframe
  className="w-full h-full"
  src={`${viewerUrl}?file=${encodeURIComponent(fileUrl)}${hash}`}
  loading="lazy"
/>

4) Ship the PDF.js viewer + worker with the build

Add scripts/copy-pdfjs.js:

import fs from "fs";
import path from "path";

const root = process.cwd();
const fromWeb = path.join(root, "node_modules", "pdfjs-dist", "web");
const fromWorker = path.join(root, "node_modules", "pdfjs-dist", "build", "pdf.worker.min.js");
const toWeb = path.join(root, "public", "pdfjs", "web");
const toBuild = path.join(root, "public", "pdfjs", "build");

fs.mkdirSync(toWeb, { recursive: true });
fs.mkdirSync(toBuild, { recursive: true });

fs.cpSync(fromWeb, toWeb, { recursive: true });
fs.copyFileSync(fromWorker, path.join(toBuild, "pdf.worker.min.js"));

console.log("Copied pdfjs viewer + worker into /public/pdfjs");


Update package.json scripts:

{
  "scripts": {
    "prepare": "node scripts/copy-pdfjs.js",
    "build": "vite build",
    "verify:pdfs": "ls -lh dist/publications || echo 'NO dist/publications'"
  }
}


Run once:

npm run prepare
git add public/pdfjs -A && git commit -m "bundle pdfjs viewer+worker"
